cmake_minimum_required(VERSION 3.20)
project(ShumiDesktopBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SHUMI_ASSERTS "Enable the existing engine assertions" ON)
get_filename_component(SHUMI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

# Compile the shared sources without altering the original project or its drivers.
set(CORE_SOURCES gameboard NNUE globals engine fen_database move_tables utility
    minimax endgameTables weights status_output)
list(TRANSFORM CORE_SOURCES PREPEND "${SHUMI_ROOT}/src/")
list(TRANSFORM CORE_SOURCES APPEND ".cpp")
add_library(shumi_webapp_core STATIC ${CORE_SOURCES})
target_include_directories(shumi_webapp_core PUBLIC "${SHUMI_ROOT}/src")
if(SHUMI_ASSERTS)
  target_compile_definitions(shumi_webapp_core PUBLIC SHUMI_FORCE_ASSERTS=1)
endif()
if(MSVC)
  target_compile_options(shumi_webapp_core PRIVATE /W3 /utf-8 $<$<CONFIG:Release>:/O2>)
endif()
find_package(Threads REQUIRED)
add_executable(shumi_bridge bridge.cpp)
target_link_libraries(shumi_bridge PRIVATE shumi_webapp_core Threads::Threads)
target_include_directories(shumi_bridge PRIVATE vendor)
target_compile_definitions(shumi_bridge PRIVATE SHUMI_REPO_ROOT="${SHUMI_ROOT}")
if(MSVC)
  target_compile_options(shumi_bridge PRIVATE /W4 /utf-8)
endif()
set_target_properties(shumi_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
